5. Frontend-to-Backend Integration Contract

The frontend communicates with the backend via the useGeminiChatStream custom hook: 

// src/hooks/useGeminiChatStream.ts
export function useGeminiChatStream() {
  const [messages, setMessages] = useState([]);
  const [isStreaming, setIsStreaming] = useState(false);

  const sendMessage = async (userPrompt: string) => {
    setIsStreaming(true);
    // Optimistically append user message
    // If backend is unavailable or NEXT_PUBLIC_MOCK_AI=true, fallback to built-in simulation engine
    const response = await fetch(`${process.env.NEXT_PUBLIC_API_URL}/api/chat`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ message: userPrompt })
    });

    const reader = response.body?.getReader();
    const decoder = new TextDecoder();
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      const chunk = decoder.decode(value);
      // parse SSE and update state token-by-token
    }
    setIsStreaming(false);
  };

  return { messages, sendMessage, isStreaming };
}
